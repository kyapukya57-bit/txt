-- ===== 敵(Humanoid)キャッシュ =====
local humans = {}
local rootCache = setmetatable({}, { __mode = "k" })

for _, d in ipairs(workspace:GetDescendants()) do
    if d:IsA("Humanoid") then humans[d] = true end
end
table.insert(conns, workspace.DescendantAdded:Connect(function(d)
    if d:IsA("Humanoid") then humans[d] = true end
end))
table.insert(conns, workspace.DescendantRemoving:Connect(function(d)
    if d:IsA("Humanoid") then humans[d] = nil; rootCache[d] = nil end
end))

local function enemyRoot(hum)
    local r = rootCache[hum]
    if r and r.Parent then return r end
    local m = hum.Parent
    r = m and (m:FindFirstChild("HumanoidRootPart") or m.PrimaryPart)
    rootCache[hum] = r
    return r
end

local function isEnemy(hum)
    local model = hum.Parent
    if not model or not model:IsA("Model") then return false end
    if model == player.Character then return false end
    if hum.Health <= 0 then return false end
    if not CONFIG.IncludePlayers and Players:GetPlayerFromCharacter(model) then
        return false
    end
    return true
end

local function tierOf(hum)
    return prioIndex[hum.Parent.Name] or math.huge
end

local function findBest(myPos)
    local best, bestTier, bestDist
    for d in pairs(humans) do
        if isEnemy(d) then
            local tier = tierOf(d)
            if tier ~= math.huge or fallback then
                local r = enemyRoot(d)
                if r then
                    local dist = (r.Position - myPos).Magnitude
                    if dist <= range and allowedTarget(r.Position) then
                        if not best or tier < bestTier
                            or (tier == bestTier and dist < bestDist) then
                            best, bestTier, bestDist = d, tier, dist
                        end
                    end
                end
            end
        end
    end
    return best, bestTier
end

local function findNearestAny(myPos)
    local best, bestDist
    for d in pairs(humans) do
        if isEnemy(d) then
            local r = enemyRoot(d)
            if r then
                local dist = (r.Position - myPos).Magnitude
                if not bestDist or dist < bestDist then
                    best, bestDist = d, dist
                end
            end
        end
    end
    return best
end

    end))
end

-- ===== 宝箱判定 =====
local function nameMatches(name)
    name = string.lower(name)
    for _, k in ipairs(keywords) do
        if k ~= "" and string.find(name, string.lower(k), 1, true) then
            return true
        end
    end
    return false
end

local function isChest(d)
    if not (d:IsA("Model") or d:IsA("BasePart")) then return false end
    if not nameMatches(d.Name) then return false end
    if d:IsA("Model") and d:FindFirstChildWhichIsA("Humanoid") then return false end
    local p = d.Parent
    while p and p ~= workspace do
        if p:IsA("Model") and nameMatches(p.Name) then return false end
        p = p.Parent
    end
    return true
end

local function objPos(d)
    if d:IsA("BasePart") then return d.Position end
    local ok, cf = pcall(function() return d:GetPivot() end)
    return ok and cf.Position or nil
end

-- ===== モデル情報の収集(検索用) =====
local function collectInfo(m, withMesh)
    local lines = {}
    local blobParts = { m.Name, m:GetFullName() }
    local hum = m:FindFirstChildWhichIsA("Humanoid")

    table.insert(lines, "名前: " .. m.Name)
    if hum then
        if hum.DisplayName ~= "" and hum.DisplayName ~= m.Name then
            table.insert(lines, "表示名: " .. hum.DisplayName)
            table.insert(blobParts, hum.DisplayName)
        end
        table.insert(lines, string.format("HP: %d / %d", hum.Health, hum.MaxHealth))
    end
    table.insert(lines, "クラス: " .. m.ClassName)
    table.insert(lines, "パス: " .. m:GetFullName())

    -- 属性(Attributes)
    local attrs = {}
    local function addAttrs(inst)
        for k, v in pairs(inst:GetAttributes()) do
            table.insert(attrs, k .. "=" .. tostring(v))
        end
    end
    addAttrs(m)
    if hum then addAttrs(hum) end
    if #attrs > 0 then
        table.insert(lines, "属性: " .. table.concat(attrs, ", "))
        table.insert(blobParts, table.concat(attrs, " "))
    end

    -- ID系の値オブジェクト
    local vals = {}
    for _, c in ipairs(m:GetChildren()) do
        if c:IsA("StringValue") or c:IsA("IntValue") or c:IsA("NumberValue") then
            table.insert(vals, c.Name .. "=" .. tostring(c.Value))
        end
    end
    if #vals > 0 then
        table.insert(lines, "値: " .. table.concat(vals, ", "))
        table.insert(blobParts, table.concat(vals, " "))
    end

    -- メッシュID(モデルID)
    if withMesh then
        local seen, meshes, scanned = {}, {}, 0
        for _, d in ipairs(m:GetDescendants()) do
            scanned += 1
            if scanned > 300 or #meshes >= 6 then break end
            local id
            if d:IsA("MeshPart") or d:IsA("SpecialMesh") then id = d.MeshId end
            if id and id ~= "" and not seen[id] then
                seen[id] = true
                table.insert(meshes, id)
            end
        end
        if #meshes > 0 then
            table.insert(lines, "メッシュID: " .. table.concat(meshes, ", "))
            table.insert(blobParts, table.concat(meshes, " "))
        end
    end

    return {
        text = table.concat(lines, "\n"),
        blob = string.lower(table.concat(blobParts, " ")),
    }
end

local function rebuildPriorityList()
    for _, c in ipairs(pList:GetChildren()) do
        if c:IsA("Frame") then c:Destroy() end
    end
    for i, name in ipairs(priority) do
        local row = newRow(pList)
        rowLabel(row, i .. ". " .. name, 140)
        rowBtn(row, "▲", 146, function()
            if i > 1 then
                priority[i], priority[i - 1] = priority[i - 1], priority[i]
                rebuildIndex(); save(); rebuildPriorityList()
            end
        end)
        rowBtn(row, "▼", 172, function()
            if i < #priority then
                priority[i], priority[i + 1] = priority[i + 1], priority[i]
                rebuildIndex(); save(); rebuildPriorityList()
            end
        end)
        rowBtn(row, "✕", 198, function()
            table.remove(priority, i)
            rebuildIndex(); save(); rebuildPriorityList()
        end)
    end
end

local function runSearch()
    for _, c in ipairs(sList:GetChildren()) do
        if c:IsA("TextButton") then c:Destroy() end
    end
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local myPos = hrp and hrp.Position or Vector3.zero
    local q = string.lower(searchBox.Text)
    local results = {}

    local function consider(m)
        local pos = objPos(m)
        if not pos then return end
        local info = collectInfo(m, searchMobsOnly)
        if q == "" or string.find(info.blob, q, 1, true) then
            table.insert(results, { model = m, info = info, dist = (pos - myPos).Magnitude })
        end
    end

    if searchMobsOnly then
        for hum in pairs(humans) do
            local m = hum.Parent
            if m and m:IsA("Model") and m ~= player.Character
                and (CONFIG.IncludePlayers or not Players:GetPlayerFromCharacter(m)) then
                consider(m)
            end
        end
    else
        for _, d in ipairs(workspace:GetDescendants()) do
            if d:IsA("Model") and d ~= player.Character
                and not Players:GetPlayerFromCharacter(d) then
                consider(d)
            end
        end
    end

    table.sort(results, function(a, b) return a.dist < b.dist end)

    for i = 1, math.min(#results, CONFIG.SearchLimit) do
        local r = results[i]
        local b = Instance.new("TextButton")
        b.Size = UDim2.new(1, -6, 0, 22)
        b.BackgroundColor3 = Color3.fromRGB(55, 55, 55)
        b.TextColor3 = Color3.new(1, 1, 1)
        b.BorderSizePixel = 0
        b.Font = Enum.Font.SourceSans
        b.TextSize = 14
        b.TextXAlignment = Enum.TextXAlignment.Left
        b.TextTruncate = Enum.TextTruncate.AtEnd
        b.Text = " " .. r.model.Name .. "  (" .. math.floor(r.dist) .. ")"
        b.Parent = sList
        b.MouseButton1Click:Connect(function()
            selectResult(r.model, r.info, r.dist)
        end)
    end

    flash(#results .. "件ヒット" .. (#results > CONFIG.SearchLimit and "(上位" .. CONFIG.SearchLimit .. "件表示)" or ""))
end

-- ===== 宝箱の出現を検知(出現しただけではTPせず、周りのモブ全滅を待つ) =====
local pending, pendingSet = {}, {}   -- 待機中の宝箱

-- 待機中の宝箱から ChestMobRadius 以内 = 宝箱の部屋(宝箱が出ていない間は部屋なし)

    end
    return false
end

-- 宝箱の周り(ChestMobRadius以内)にいるモブの数
local function mobsNear(pos)
    local n = 0
    for d in pairs(humans) do
        if isEnemy(d) then
            local r = enemyRoot(d)
            if r and (r.Position - pos).Magnitude <= CONFIG.ChestMobRadius then
                n += 1
            end
        end
    end
    return n
end

local function isLocked(obj)
    local v = obj:GetAttribute(CONFIG.LockAttr)
    return v ~= nil and string.lower(tostring(v)) == string.lower(CONFIG.LockValue)
end

local function isMine(obj)
    if not CONFIG.OwnerCheck then return true end
    local o = obj:GetAttribute("Owner")
    return o == nil or tonumber(o) == player.UserId
end

-- 戻り値: "drop"(もう扱わない) / "go"(今TPする) / nil(待機), 表示用テキスト
local function evaluate(e, now)
    local d = e.obj
    if not d.Parent then return "drop" end                       -- 宝箱が消えた(開いた)
    if now - e.t0 > CONFIG.ChestTimeout then return "drop" end   -- 諦め
    if not isMine(d) then return "drop" end                      -- 他人の宝箱

    local pos = objPos(d)
    if not pos then return nil, "準備中" end

    -- TP後の確認(開いたか?)
    if e.nextTry then
        if now < e.nextTry then return nil, "開錠確認中" end
        if not CONFIG.WaitForClear or not isLocked(d) or e.tries >= CONFIG.MaxRetries then
            return "drop"
        end
        e.nextTry, e.clearSince = nil, nil   -- まだ施錠 → もう一度待ってから再挑戦
    end

    if CONFIG.WaitForClear then
        local n = mobsNear(pos)
        if n > 0 then
            e.seenMob, e.clearSince, e.tries = true, nil, 0
            return nil, "モブ" .. n .. "体待ち"
        end
        -- 宝箱が先に出てモブが後からわくので、一度もモブを見ていない間は猶予を持たせる
        if not e.seenMob and now - e.t0 < CONFIG.SpawnGrace then
            return nil, "モブ出現待ち"
        end
        e.clearSince = e.clearSince or now
        local clear = now - e.clearSince
        if clear < CONFIG.ClearConfirm then return nil, "全滅確認中" end
        -- モブ0でもまだ施錠中なら、少しだけ解錠を待つ
        if isLocked(d) and clear < CONFIG.ClearConfirm + CONFIG.LockFallback then
            return nil, "解錠待ち"
        end
    end

    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil, "キャラ待ち" end
    if (pos - hrp.Position).Magnitude > CONFIG.MaxDistance then return nil, "宝箱が遠い" end
    return "go"
end

-- 宝箱を開ける(敵追跡は busy の間止まる)
-- walk: 手前にTPして宝箱へ歩く(ゲームが「歩いて開ける」判定のため) / direct: 宝箱へ直接TP
local function tpUnderChest(d)
    busy = true
    status.Text = "宝箱へ移動中..."
    task.wait(0.1)   -- ノークリップが効くのを待つ

    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local pos = d.Parent and objPos(d)
    if pos and hrp then
        local origin = hrp.CFrame
        local rot = origin - origin.Position

        if CONFIG.ChestOpenMode == "walk" and hum then
            walking = true
            pcall(function()
                local dir = Vector3.new(origin.Position.X - pos.X, 0, origin.Position.Z - pos.Z)
                dir = dir.Magnitude > 1 and dir.Unit or Vector3.new(1, 0, 0)
                local from = pos + dir * CONFIG.ChestApproachDist + Vector3.new(0, 3, 0)
                hrp.Anchored = false
                hrp.CFrame = CFrame.new(from) * rot
                hrp.AssemblyLinearVelocity = Vector3.zero
                local t0 = os.clock()
                while os.clock() - t0 < CONFIG.ChestOpenWait do
                    if not d.Parent then break end
                    if os.clock() - t0 > 0.5 and d:GetAttribute(CONFIG.LockAttr) ~= nil
                        and not isLocked(d) then break end
                    hum:MoveTo(pos)
                    task.wait(0.2)
                end
                task.wait(0.2)
            end)
            walking = false
        else
            pcall(function()
                hrp.CFrame = CFrame.new(pos + Vector3.new(0, CONFIG.ChestOpenOffsetY, 0)) * rot
                hrp.AssemblyLinearVelocity = Vector3.zero
                hrp.Anchored = true
                task.wait(chestStay)
            end)
        end

        if hrp.Parent then
            hrp.CFrame = origin
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.Anchored = false
        end
    end
    busy = false
    refresh()
end

-- ===== 武器の持ち替え(キー入力を送る。同じキーを2回押すと外れるので、持っている道具を見て判断) =====
local updateWeapon
do
    local VIMw
    pcall(function() VIMw = game:GetService("VirtualInputManager") end)
    local learned = {}      -- キー -> そのキーで持てた道具の名前(最初に押したとき覚える)
    local current = "pick"  -- 今の状態(最初はつるはしのつもりで、何もしない)
    local working = false

    local function equippedTool()
        local c = player.Character
        return c and c:FindFirstChildOfClass("Tool")
    end
    local function press(key)
        VIMw:SendKeyEvent(true, key, false, game)
        task.wait(0.05)
        VIMw:SendKeyEvent(false, key, false, game)
    end
    local function equip(key)
        local known = learned[key]
        local cur = equippedTool()
        if cur and known and cur.Name == known then return end   -- もう持っている
        press(key)
        -- 反映は遅れることがあるので、変化するまで最大0.8秒待つ(待つだけで、勝手に再度は押さない)
        local t0, now = os.clock(), cur
        repeat
            task.wait(0.1)
            now = equippedTool()
        until now ~= cur or os.clock() - t0 > 0.8
        -- 「持っていたのに空になった」= 同じ道具を持っていて外れた場合だけ、1回だけ押し直す
        -- (何も持っていない状態から押して変化が見えないときは、押し直さない=素手にしない)
        if cur and not now then
            press(key)
            t0 = os.clock()
            repeat
                task.wait(0.1)
                now = equippedTool()
            until now or os.clock() - t0 > 0.8
        end
        if now and not known then learned[key] = now.Name end
    end

    -- want: "sword" か "pick"。状態が変わったときだけキーを押す
    function updateWeapon(want)
        if not CONFIG.WeaponSwap or not VIMw or working or want == current then return end
        working = true
        task.spawn(function()
            equip(want == "sword" and CONFIG.SwordKey or CONFIG.PickaxeKey)
            current = want
            working = false
        end)
    end
end

-- 待機中の宝箱を監視(待っている間は敵追跡は止めない)
task.spawn(function()
    while alive do
        task.wait(0.2)
        if not (chestOn and autoOn) then   -- 宝箱を開けるのは自動周回中だけ
            if #pending > 0 then pending, pendingSet = {}, {} end
            chestInfo = ""
        else
            local i, shown = 1, nil
            while i <= #pending do
                local e = pending[i]
                local act, why = evaluate(e, os.clock())
                if act == "drop" then
                    if autoOn and cardEnabled() and (e.tries > 0 or not e.obj.Parent) then
                        cardDeadline = os.clock() + CONFIG.CardWait
                        cardPicks = 0
                    end
                    pendingSet[e.obj] = nil
                    table.remove(pending, i)
                elseif act == "go" then
                    e.tries += 1
                    tpUnderChest(e.obj)
                    e.nextTry = os.clock() + 0.7   -- 解錠状態が反映されるのを待って確認
                    i += 1
                else
                    shown = shown or why
                    i += 1
                end
            end
            if shown and #pending > 1 then shown = shown .. " 他" .. (#pending - 1) end
            chestInfo = shown or ""
        end
        -- 宝箱フェーズ(待機中の宝箱がある間)だけ剣、終わったらつるはし
        updateWeapon((chestOn and autoOn and #pending > 0) and "sword" or "pick")
        if not busy and msgToken == 0 then status.Text = statusLine() end
    end
end)

local lastScan = 0
table.insert(conns, RunService.Heartbeat:Connect(function()
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    if CONFIG.ShowRange then sphere.CFrame = CFrame.new(hrp.Position) end
    if not enemyActive() or busy or oreWorking then return end

    local valid = false
    if target and target.Parent and target.Health > 0 then
        local r = enemyRoot(target)
        if r and (r.Position - hrp.Position).Magnitude <= range + enemyDepth + 5
            and allowedTarget(r.Position) then
            valid = true
        end
    end
    if not valid then target = nil end

    local now = os.clock()
    if now - lastScan > CONFIG.ScanInterval then
        lastScan = now
        local best, bestTier = findBest(hrp.Position)
        if best and not (target and tierOf(target) == bestTier) then
            target = best
        end
    end

    if target then
        local r = enemyRoot(target)
        if r then
            local pos = r.Position - Vector3.new(0, enemyDepth, 0)
            local rot = hrp.CFrame - hrp.CFrame.Position
            hrp.CFrame = CFrame.new(pos) * rot
            hrp.AssemblyLinearVelocity = Vector3.zero
        end
    end
end))

    -- ===== 周辺モデルの追跡(検索モードON中だけ) =====
    local allModels, modelConns, nearList = {}, {}, {}

    local function startTracking()
        allModels = {}
        for _, d in ipairs(workspace:GetDescendants()) do
            if d:IsA("Model") then allModels[d] = true end
        end
        modelConns[1] = workspace.DescendantAdded:Connect(function(d)
            if d:IsA("Model") then allModels[d] = true end
        end)
        modelConns[2] = workspace.DescendantRemoving:Connect(function(d)
            allModels[d] = nil
        end)
    end

    local function stopTracking()
        for _, c in ipairs(modelConns) do c:Disconnect() end
        modelConns, allModels, nearList = {}, {}, {}
    end

    local function refreshNear(myPos)
        local list = {}
        for m in pairs(allModels) do
            if m.Parent and m ~= player.Character
                and not m:FindFirstChildWhichIsA("Humanoid") then
                local ok, pcf = pcall(function() return m:GetPivot() end)
                if ok and (pcf.Position - myPos).Magnitude < NEAR_RADIUS then
                    local ok2, cf, size = pcall(function() return m:GetBoundingBox() end)
                    if ok2 and size.Magnitude < BIG_SIZE then
                        table.insert(list, { obj = m, pos = cf.Position })
                    end
                end
            end
        end
        nearList = list
    end

    -- ===== 候補の判定 =====
    local function isBig(obj)
        local s = sizeCache[obj]
        if s == nil then
            if obj:IsA("Model") then
                local ok, ext = pcall(function() return obj:GetExtentsSize() end)
                s = ok and ext.Magnitude or 0
            elseif obj:IsA("BasePart") then
                s = obj.Size.Magnitude
            else
                s = 0
            end
            sizeCache[obj] = s
        end
        return s > BIG_SIZE
    end

    -- 小さいほど優先: 0=モブ 1=宝箱キーワード 3=普通 9=大きい/床など
    local function scoreOf(obj)
        if obj:IsA("Model") and obj:FindFirstChildWhichIsA("Humanoid") then return 0 end
        if nameMatches(obj.Name) then return 1 end
        local lname = string.lower(obj.Name)
        for _, w in ipairs(IGNORE_NAMES) do
            if string.find(lname, w, 1, true) then return 9 end
        end
        if isBig(obj) then return 9 end
        return 3
    end

    local function pickObj(part)
        local nearest
        local cur = part.Parent
        while cur and cur ~= workspace do
            if cur:IsA("Model") then
                if cur:FindFirstChildWhichIsA("Humanoid") then return cur end
                nearest = nearest or cur
            end
            cur = cur.Parent
        end
        return nearest or part
    end

    local rp = RaycastParams.new()
    rp.FilterType = Enum.RaycastFilterType.Exclude

    local function projectPx(cam, pos, loc)
        local v, onScreen = cam:WorldToViewportPoint(pos)
        if not onScreen or v.Z <= 0 then return nil end
        return (Vector2.new(v.X, v.Y) - loc).Magnitude
    end

    local function buildCandidates(cam, loc)
        local cands, seen = {}, {}

        local function add(obj, part, px)
            if not obj or obj == player.Character then return end
            local c = seen[obj]
            if c then
                if px < c.px then c.px = px end
                return
            end
            c = { obj = obj, part = part, score = scoreOf(obj), px = px, idx = #cands + 1 }
            seen[obj] = c
            table.insert(cands, c)
        end

        -- 1) 貫通レイキャスト(手前の柱や床で止まらない)
        local filter = { sphere }
        if player.Character then table.insert(filter, player.Character) end
        rp.FilterDescendantsInstances = filter
        local ray = cam:ViewportPointToRay(loc.X, loc.Y)
        for _ = 1, 12 do
            local res = workspace:Raycast(ray.Origin, ray.Direction * 1000, rp)
            if not res or not res.Instance then break end
            local part = res.Instance
            local obj = pickObj(part)
            add(obj, part, 0)
            table.insert(filter, obj)   -- そのモデルごと除外して奥を探す
            rp.FilterDescendantsInstances = filter
        end

        -- 2) 画面上の位置で探す(当たり判定の無いモデルも拾う)
        for hum in pairs(humans) do
            local m = hum.Parent
            local r = m and enemyRoot(hum)
            if r and m ~= player.Character then
                local px = projectPx(cam, r.Position, loc)
                if px and px < PIXEL_RADIUS * 1.5 then add(m, r, px) end
            end
        end
        for _, e in ipairs(nearList) do
            if e.obj.Parent then
                local px = projectPx(cam, e.pos, loc)
                if px and px < PIXEL_RADIUS then add(e.obj, nil, px) end
            end
        end

        table.sort(cands, function(a, b)
            if a.score ~= b.score then return a.score < b.score end
            if math.abs(a.px - b.px) > 0.5 then return a.px < b.px end
            return a.idx < b.idx
        end)
        return cands
    end

    local function partMesh(part)
        if not part then return nil end
        if part:IsA("MeshPart") and part.MeshId ~= "" then return part.MeshId end
        local sm = part:FindFirstChildWhichIsA("SpecialMesh")
        if sm and sm.MeshId ~= "" then return sm.MeshId end
        return nil
    end

    local TAGS = { [0] = "[モブ]", [1] = "[宝箱?]", [9] = "[大/床]" }

    local function inspect()
        local loc = UIS:GetMouseLocation()
        local cam = workspace.CurrentCamera
        if not cam then return end

        local cands = buildCandidates(cam, loc)
        if #cands == 0 then
            tip.Visible = false
            hl.Adornee = nil
            lastTop = nil
            return
        end

        -- 一番上の候補が変わったら選択をリセット
        if cands[1].obj ~= lastTop then
            lastTop = cands[1].obj
            candIdx, level = 1, 0
        end
        if candIdx > #cands then candIdx = 1 end

        local c = cands[candIdx]
        local obj = c.obj

        -- Zキーで親モデルへ
        for _ = 1, level do
            local p = obj.Parent
            while p and p ~= workspace and not p:IsA("Model") do p = p.Parent end
            if p and p ~= workspace then obj = p else break end
        end

        local text
        if obj:IsA("Model") then
            local info = infoCache[obj]
            if not info then
                info = collectInfo(obj, true)
                infoCache[obj] = info
            end
            local isMob = obj:FindFirstChildWhichIsA("Humanoid") ~= nil
            text = (isMob and "【モブ】" or "【モデル】") .. "\n" .. info.text
        else
            text = "【パーツ】\n名前: " .. obj.Name .. "\nクラス: " .. obj.ClassName
        end

        local mesh = partMesh(c.part) or (obj:IsA("BasePart") and partMesh(obj))
        if c.part then
            text = text .. "\nヒットパーツ: " .. c.part.Name .. (mesh and (" / " .. mesh) or "")
        elseif mesh then
            text = text .. "\nメッシュID: " .. mesh
        end

        -- 候補一覧
        text = text .. "\n--- 候補 " .. candIdx .. "/" .. #cands .. " ---"
        for i = 1, math.min(#cands, 6) do
            local cc = cands[i]
            text = text .. "\n" .. (i == candIdx and "▶ " or "   ") .. cc.obj.Name .. " " .. (TAGS[cc.score] or "")
        end
        text = text .. "\n[" .. NEXT_KEY.Name .. ":次の候補] [" .. UP_KEY.Name
            .. ":親へ] [" .. COPY_KEY.Name .. ":コピー]"

        hl.Adornee = obj
        lastText = text
        tip.Text = text
        tip.Visible = true

        local vp = cam.ViewportSize
        local x = math.min(loc.X + 16, vp.X - 330)
        local y = math.min(loc.Y + 16, vp.Y - tip.AbsoluteSize.Y - 10)
        tip.Position = UDim2.fromOffset(math.max(0, x), math.max(0, y))
    end

    -- ===== ON/OFF =====
    local function updateBtn()
        inspectBtn.Text = "検索モード(ホバーでID表示): " .. (inspectOn and "ON" or "OFF")
            .. " [" .. INSPECT_KEY.Name .. "]"
        inspectBtn.BackgroundColor3 = inspectOn and Color3.fromRGB(40, 120, 60) or Color3.fromRGB(60, 60, 60)
    end
    updateBtn()

    local function toggleInspect()
        inspectOn = not inspectOn
        if inspectOn then
            startTracking()
            lastCheck = 0
        else
            stopTracking()
            tip.Visible = false
            hl.Adornee = nil
            lastTop, candIdx, level = nil, 1, 0
        end
        updateBtn()
        flash("検索モード " .. (inspectOn and "ON" or "OFF"))
    end
    inspectBtn.MouseButton1Click:Connect(toggleInspect)

    local lastNear = 0
    table.insert(conns, RunService.Heartbeat:Connect(function()
        if not inspectOn then return end
        local now = os.clock()
        if now - lastCheck < 0.05 then return end
        lastCheck = now

        local char = player.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if hrp and now - lastNear > 1 then
            lastNear = now
            refreshNear(hrp.Position)
        end
        inspect()
    end))

    table.insert(conns, UIS.InputBegan:Connect(function(input, gp)
        if gp then return end
        if input.KeyCode == INSPECT_KEY then
            toggleInspect()
        elseif inspectOn and input.KeyCode == COPY_KEY then
            if lastText ~= "" then
                local ok = pcall(function() setclipboard(lastText) end)
                flash(ok and "コピーしました" or "この環境ではコピー不可")
            end
        elseif inspectOn and input.KeyCode == UP_KEY then
            level += 1
        elseif inspectOn and input.KeyCode == NEXT_KEY then
            candIdx += 1
            level = 0
        end
    end))

    -- 後始末も一緒に行うように包む
    if getgenv then
        local oldCleanup = getgenv().__UnderTP
        getgenv().__UnderTP = function()
            stopTracking()
            tipGui:Destroy()
            if oldCleanup then oldCleanup() end
        end
    end
end

    local function cut(s, n)   -- 文字数で切る(日本語対応)
        if utf8.len(s) and utf8.len(s) > n then
            return string.sub(s, 1, utf8.offset(s, n + 1) - 1) .. "…"
        end

    end

    local function cleanText(s)
        s = s:gsub("<[^>]+>", ""):gsub("%s+", " ")
        return s:match("^%s*(.-)%s*$")
    end

        for _, d in ipairs(o:GetDescendants()) do

            if #out >= limit or scanned > 300 then break end
            consider(d)
        end
        return out

                    local rel = cf:PointToObjectSpace(pp)
                    local inside = math.abs(rel.X) <= size.X / 2 and math.abs(rel.Z) <= size.Z / 2
                    local dist = Vector3.new(cf.Position.X - pp.X, 0, cf.Position.Z - pp.Z).Magnitude
                    local score = (inside and 0 or 1e6) + dist
                    if not bestScore or score < bestScore then
                        best, bestScore = { obj = d, cf = cf, size = size }, score
                    end
                end
            end
        end
        if not best then
            floorCache.info = nil
            return nil
        end
        local cf, size = best.cf, best.size
        local axis, len
        if size.X >= size.Z then axis, len = cf.RightVector, size.X else axis, len = cf.LookVector, size.Z end
        axis = Vector3.new(axis.X, 0, axis.Z)
        axis = axis.Magnitude > 0.01 and axis.Unit or Vector3.new(1, 0, 0)
        -- 最初は「今いる側から遠いほうの端」へ向かう。違えば反転ボタンで逆にする
        local sign = dirSign[best.obj]
        if not sign then
            sign = (pp - cf.Position):Dot(axis) > 0 and -1 or 1
            dirSign[best.obj] = sign
        end
        local info = { top = cf.Position.Y + size.Y / 2, center = cf.Position, axis = axis, len = len, sign = sign }
        floorCache.info = info
        return info
    end

        end
        return info.axis * info.sign * (flip and -1 or 1)
    end

    -- 鉱石か: 自分の名前、または親(フォルダなど)の名前が登録名に一致するモデル/パーツ
    local function isOre(d)
        if #oreKeywords == 0 then return false end
        if not (d:IsA("Model") or d:IsA("BasePart")) then return false end
        local par = d.Parent
        if not (oreNameMatches(d.Name) or (par and par ~= workspace and oreNameMatches(par.Name))) then

        if d:IsA("Model") and d:FindFirstChildWhichIsA("Humanoid") then return false end
        local p = par
        while p and p ~= workspace do   -- 鉱石モデルの中のパーツは、モデル1個として数える
            if p:IsA("Model") and oreNameMatches(p.Name) then return false end
            p = p.Parent
        end
        return true
    end

    -- 壁に当たった場所(origin)の周りの鉱石へ順にTPして掘り、なくなったら origin へ戻る
    local function oreRun(origin)
        oreWorking = true
        local char = player.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local moved, anchored = false, false
        local skipped, n = {}, 0
        pcall(function()
            while hrp and hrp.Parent and autoOn and oreOn and not busy and #pending == 0 do
                -- 今いる場所から一番近い、まだ残っている鉱石
                local best, bestD
                for d in pairs(oreSet) do
                    if d.Parent and not skipped[d] then
                        local p = objPos(d)
                        if p and (p - origin).Magnitude <= CONFIG.OreRadius then
                            local dd = (p - hrp.Position).Magnitude
                            if not bestD or dd < bestD then best, bestD = d, dd end
                        end
                    end
                end
                if not best then break end   -- 周りに鉱石がもうない
                local pos = objPos(best)
                if not pos then
                    skipped[best] = true
                else
                    n += 1
                    moved, anchored = true, true
                    autoInfo = "自動: 鉱石を掘る(" .. n .. "個目)"
                    local rot = hrp.CFrame - hrp.CFrame.Position
                    hrp.Anchored = true
                    hrp.CFrame = CFrame.new(pos + Vector3.new(0, CONFIG.OreOffsetY, 0)) * rot
                    hrp.AssemblyLinearVelocity = Vector3.zero
                    local t0 = os.clock()
                    while best.Parent and hrp.Parent and autoOn and not busy and #pending == 0
                        and os.clock() - t0 < CONFIG.OreStay do
                        task.wait(0.1)
                    end
                    if best.Parent then skipped[best] = true end   -- 掘れなかった鉱石は飛ばす
                end
                task.wait()
            end
        end)
        -- 宝箱の処理(busy)に入ったら位置には触らない。それ以外は壁へ戻る
        if hrp and hrp.Parent and not busy then
            if moved and autoOn then
                local rot = hrp.CFrame - hrp.CFrame.Position
                hrp.CFrame = CFrame.new(origin) * rot
                hrp.AssemblyLinearVelocity = Vector3.zero
            end
            if anchored then hrp.Anchored = false end
        end
        nextOreCheck = os.clock() + CONFIG.OreRecheck
        autoInfo = ""
        oreWorking = false
    end

    local function findDepthLabel()
        local kw = string.lower(CONFIG.DepthKeyword)
        local numLabel
        for _, d in ipairs(pg:GetDescendants()) do
            if d:IsA("TextLabel") then
                local sg = d:FindFirstAncestorWhichIsA("ScreenGui")
                if sg and not OWN[sg.Name] then
                    local t = string.lower(cleanText(d.Text))
                    if string.find(t, kw, 1, true) and string.match(t, "%d") then return d end
                    if not numLabel and string.match(t, "^[%d,%.]+%s*m$") and d.Parent then
                        for _, sib in ipairs(d.Parent:GetChildren()) do
                            if sib ~= d and sib:IsA("TextLabel")
                                and string.find(string.lower(cleanText(sib.Text)), kw, 1, true) then
                                numLabel = d
                                break
                            end
                        end
                    end
                end
            end
        end
        return numLabel
    end

    local function readDepth()
        if not (depthLabel and depthLabel.Parent) then
            local now = os.clock()
            if not autoOn or now - lastDepthScan < 3 then return nil end   -- 探すのは自動周回中だけ、3秒おき
            lastDepthScan = now
            depthLabel = findDepthLabel()
            if not depthLabel then return nil end
        end
        local num = string.match(cleanText(depthLabel.Text), "(%d[%d,%.]*)")
        return num and tonumber((num:gsub(",", ""))) or nil
    end

                stopped = true
                autoInfo = ""          -- 宝箱の状況は chestInfo が表示する
            elseif os.clock() < cardDeadline then
                stopped = true
                autoInfo = "自動: カード選択待ち"
            end

            local pos = hrp.Position
            if not stageStart then stageStart = pos end   -- 最初は開始時の位置(深さが変わったら更新)
            local dirv = advanceDir(info)

                local t = os.clock()

                    stallT, stallPos = t, pos

                    if (pos - stallPos):Dot(dirv) < speed * (t - stallT) * 0.2 then stalled = true end
                    stallT, stallPos = t, pos

                autoInfo = "自動: 床の端に到着"

            if arrived then

                sweepCount += 1

                floorCache.t = 0   -- 戻った先の床の情報を取り直す

            hrp.CFrame = CFrame.new(np) * (hrp.CFrame - hrp.CFrame.Position)
            hrp.AssemblyLinearVelocity = Vector3.zero
            advancing = true
        end

        if not inAdvance then blockedSince = nil end
        local now = os.clock()
        if now - lastStatus > 0.25 then
            lastStatus = now
            if not busy and msgToken == 0 then status.Text = statusLine() end
        end
    end))

    local function hits()
        local m = UIS:GetMouseLocation()
        local inset = GuiService:GetGuiInset()
        local res, seen = {}, {}
        -- 通常のScreenGuiは(マウス-inset)、IgnoreGuiInsetのものはマウス座標で判定される
        local function add(list, ignoreInset)
            for _, o in ipairs(list) do
                if not seen[o] then
                    local sg = o:FindFirstAncestorWhichIsA("ScreenGui")
                    if sg and not OWN[sg.Name] and sg.IgnoreGuiInset == ignoreInset then
                        seen[o] = true
                        table.insert(res, { obj = o, sg = sg, i = #res + 1 })
                    end
                end
            end
        end
        add(pg:GetGuiObjectsAtPosition(m.X - inset.X, m.Y - inset.Y), false)
        add(pg:GetGuiObjectsAtPosition(m.X, m.Y), true)

        local out = {}
        for _, h in ipairs(res) do
            h.score = h.obj:IsA("GuiButton") and 0 or (hasInput(h.obj) and 1 or 2)
            if h.score < 2 or CONFIG.GuiShowAll then table.insert(out, h) end
        end
        table.sort(out, function(a, b)
            if a.score ~= b.score then return a.score < b.score end
            if a.sg.DisplayOrder ~= b.sg.DisplayOrder then return a.sg.DisplayOrder > b.sg.DisplayOrder end
            return a.i < b.i
        end)
        return out
    end

    local function describe(o)
        local lines = {}
        table.insert(lines, (o:IsA("GuiButton") and "【ボタン】 " or "【GUI】 ") .. o.ClassName)

        table.insert(lines, "パス: " .. o:GetFullName())

        while c and c ~= pg do
            table.insert(chain, 1, c.Name)
            c = c.Parent
        end
        local expr = 'game:GetService("Players").LocalPlayer.PlayerGui'
        for _, n in ipairs(chain) do
            expr = expr .. ":FindFirstChild(" .. string.format("%q", n) .. ")"
        end
        table.insert(lines, "取得: " .. expr)

        local par = o.Parent
        if par then
            local idx, same, sameIdx = 0, 0, 0
            for i, ch in ipairs(par:GetChildren()) do
                if ch == o then idx = i end
                if ch.Name == o.Name then
                    same += 1
                    if ch == o then sameIdx = same end
                end
            end
            table.insert(lines, string.format("親の子の%d番目(同名 %d/%d)", idx, sameIdx, same))
        end

        local own = textsOf(o, 8)
        if #own > 0 then table.insert(lines, "文字: " .. table.concat(own, " | ")) end
        local root = cardRoot(o)
        if root ~= o then
            local rt = textsOf(root, 10)
            if #rt > 0 then
                table.insert(lines, "カード全体の文字(" .. root.Name .. "): " .. table.concat(rt, " | "))
            end
        end
        local im = imagesOf(root)
        if #im > 0 then table.insert(lines, "画像: " .. table.concat(im, ", ")) end

        table.insert(lines, string.format("位置: %d,%d  サイズ: %d x %d",
            math.floor(o.AbsolutePosition.X), math.floor(o.AbsolutePosition.Y),
            math.floor(o.AbsoluteSize.X), math.floor(o.AbsoluteSize.Y)))

        local attrs = {}
        for k, v in pairs(o:GetAttributes()) do table.insert(attrs, k .. "=" .. tostring(v)) end
        if #attrs > 0 then table.insert(lines, "属性: " .. table.concat(attrs, ", ")) end
        return table.concat(lines, "\n")
    end

        end
        updateGuiBtn()
        flash("GUIボタン検索 " .. (guiOn and "ON" or "OFF"))
    end
    guiBtn.MouseButton1Click:Connect(toggleGui)

    table.insert(conns, RunService.Heartbeat:Connect(function()
        if not guiOn then return end
        local now = os.clock()
        if now - lastCheck < 0.08 then return end
        lastCheck = now
        inspectGui()
    end))

    table.insert(conns, UIS.InputBegan:Connect(function(input, gp)
        if gp then return end
        local k = input.KeyCode
        if k == CONFIG.AutoKey then
            toggleAuto()
        elseif k == GUI_KEY then
            toggleGui()
        elseif guiOn and k == COPY_KEY then
            if lastText ~= "" then
                local ok = pcall(function() setclipboard(lastText) end)
                flash(ok and "コピーしました" or "この環境ではコピー不可")
            end
        elseif guiOn and k == NEXT_KEY then
            candIdx += 1
        elseif guiOn and k == CARD_KEY then
            if lastObj then
                local s = textsOf(cardRoot(lastObj), 1)[1]
                if s then
                    cardBox.Text = s
                    flash("入力欄に: " .. cut(s, 10))
                else
                    flash("文字が見つかりません")
                end
            end
        end
    end))

    -- 後始末
    if getgenv then
        local oldCleanup = getgenv().__UnderTP
        getgenv().__UnderTP = function()
            autoOn = false
            gtipGui:Destroy()
            if oldCleanup then oldCleanup() end
        end
    end

    rebuildCardList()
    updateFlipBtn()
    updateAutoBtn()
    updateGuiBtn()
end

refresh()
rebuildPriorityList()
rebuildKeywordList()
